cmake_minimum_required(VERSION 3.22)
project(hyperspectrum_tap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "Only arm64-v8a is validated")
    endif()
    add_library(hyperspectrum_tap SHARED module.cpp companion.cpp ui_bridge.cpp engine.cpp spectrum_fft.cpp spatial_renderer.cpp)
    string(LENGTH "${TAP_SOURCE_BUILD}" TAP_SOURCE_BUILD_LENGTH)
    if(NOT TAP_SOURCE_BUILD_LENGTH EQUAL 64 OR NOT TAP_SOURCE_BUILD MATCHES "^[0-9a-f]+$")
        message(FATAL_ERROR "Build through build.py to record the source digest")
    endif()
    target_compile_definitions(hyperspectrum_tap PRIVATE TAP_SOURCE_BUILD="${TAP_SOURCE_BUILD}")
    target_compile_options(hyperspectrum_tap PRIVATE
        -Wall -Wextra -Werror -fno-exceptions -fno-rtti
        -fvisibility=hidden -fvisibility-inlines-hidden -ffunction-sections -fdata-sections
        -mbranch-protection=standard)
    target_link_options(hyperspectrum_tap PRIVATE
        "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
        -Wl,--gc-sections -Wl,--no-undefined -Wl,-z,relro -Wl,-z,now)
    target_link_libraries(hyperspectrum_tap PRIVATE dl)
else()
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(engine_test engine.cpp spectrum_fft.cpp spatial_renderer.cpp engine_test.cpp)
    target_compile_options(engine_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(engine_test PRIVATE Threads::Threads)
    if(TAP_SANITIZE)
        target_compile_options(engine_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(engine_test PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME engine_behavior COMMAND engine_test)
endif()
